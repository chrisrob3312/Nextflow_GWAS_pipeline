#    Leave a line out, or comment it, to get the built-in default (bin/run_record.sh, RR_SWITCHES).

#  RULES
#    * An unknown or misspelled name, a name given twice, or a value outside what the setting
#      allows STOPS the launch, naming the line. Nothing here is ever silently ignored.
#    * PRECEDENCE: built-in default < this file < arm.<label> (when ARM=<label>) < an environment
#      variable of the same name, so a one-off override needs no edit:
#          TIME_ORIGIN=diagnosis RUN_LABEL=from_dx slurm/submit_full_pipeline.sh
#    * The run record's params.tsv says, per switch, where its value came from:
#      default / settings / arm:<label> / environment (and bundle / job for the derived ones).
#    * Site settings (SLURM_PARTITION, SLURM_ACCOUNT, LOG_DIR) stay in the environment

## Traits ---------------------------------------------------------------------
# trait.<name> = binary | quantitative | survival[:<time column>:<event column>]
#   <name> is the phenotype column (binary 0/1, quantitative numeric). A survival trait reads
#   <name>_time and <name>_event unless the columns are given. TRAITS_LIST defaults to every
#   trait declared here, in this order; set TRAITS_LIST (here or in the environment) to run fewer.
trait.mrd       = binary          # end-of-induction MRD >= 0.01 % (0/1)
trait.mrd_level = quantitative    # end-of-induction MRD level 1-4, tested as one linear trend
trait.efs       = survival        # efs_time / efs_event, days from diagnosis
trait.os        = survival        # os_time / os_event, days from diagnosis

## Time origin of the survival traits -----------------------------------------
# TIME_ORIGIN: eoi = count time from the landmark in EOI_COL (time - EOI_COL; a sample whose event
#   or censoring is at or before it, or without it, is left out of that trait and counted);
#   diagnosis = the times as shipped. Applies to the GWAS, PRS validation and GxG.  [eoi | diagnosis]
TIME_ORIGIN = eoi
# EOI_COL: the landmark column, days from diagnosis to the end of induction.  [a column name]
EOI_COL = eoi_day

## Covariates -----------------------------------------------------------------
# COVARIATES: the model covariates. Unset = the bundle's phenotypes/covariates.txt (this study:
#   sex, age_at_diagnosis, wbc_dx_log10, cyto_risk), then PC1..PC<N_PCS_MODEL>.  [column names]
# COVARIATES = sex,age_at_diagnosis,wbc_dx_log10,cyto_risk,PC1,PC2,PC3,PC4,PC5
# N_PCS_MODEL: PCs in the default covariates; 5-8 for this cohort: choose from the PC-AiR scree plot
#   and the genomic-control lambda of a first run (docs/WORKFLOW.md, "How many PCs").  [whole number >= 1]
N_PCS_MODEL = 5
# FACTOR_COVARIATES: categorical covariates, column:reference_level; each enters the model as 0/1
#   indicators for its other levels (added to COVARIATES if not named there). The reference may be
#   @largest: the level with the most samples among those analysed (ties: the first alphabetically),
#   named in every results header.  [column:level or column:@largest,...]
#   This study (decided 18 Nov 2026): cytogenetic risk favorable / unfavorable / neutral, reference
#   favorable; an unknown category is coded neutral.
FACTOR_COVARIATES = cyto_risk:favorable
# COX_EXTRA_COVARIATES: covariates for the survival traits only. mrd_level as ONE linear trend term,
#   in the whole-cohort analysis; the MRD arms below set it empty.  [column names, may be empty]
COX_EXTRA_COVARIATES = mrd_level
# COX_STRATA: a column whose levels each get their own Cox baseline hazard; empty = none.  [column name or empty]
COX_STRATA =
# REF_ANCESTRY: reference ancestry of the Tractor local-ancestry terms.  [an ancestry label]
REF_ANCESTRY = EUR

## Strata and filters ---------------------------------------------------------
# MIN_STRATUM_N: smallest stratum analysed (decided 10 Nov 2026).  [whole number >= 1]
MIN_STRATUM_N = 50
# MAC_MIN: minor allele count an ancestry's dose term needs to be tested (open: the contract says 20).  [whole number >= 1]
MAC_MIN = 10
# PC_SCALE: the PCs a stratum's GWAS adjusts for; POOLED always uses the cohort PCs.  [stratum | cohort]
PC_SCALE = stratum
# MIX_RETAIN: analyse the bundle's MIX stratum.  [true | false]
MIX_RETAIN = true

## Subset arms (overall, MRD-positive, MRD-negative) ----------------------------
# The whole-cohort run (no ARM) adjusts EFS and OS for mrd_level. Each MRD arm runs EFS and OS
# only, in one MRD group, without an MRD covariate:
#     slurm/submit_full_pipeline.sh                  # whole cohort, every ancestry stratum
#     ARM=mrd_pos slurm/submit_full_pipeline.sh      # results/mrd_pos/
#     ARM=mrd_neg slurm/submit_full_pipeline.sh      # results/mrd_neg/
#     ARM=all slurm/submit_full_pipeline.sh          # the three above (ARM_ALL), one launch each
# arm.<label> = SWITCH=value ...: the switches an arm sets (they beat this file, lose to the environment).
arm.mrd_pos = SAMPLE_SUBSET=mrd=1 TRAITS_LIST=efs,os COX_EXTRA_COVARIATES=
arm.mrd_neg = SAMPLE_SUBSET=mrd=0 TRAITS_LIST=efs,os COX_EXTRA_COVARIATES=
# ARM_ALL: the arms ARM=all runs after the whole cohort, in order (empty: every arm below).  [arm labels]
ARM_ALL = mrd_pos,mrd_neg
# SUBSET_LINKED_COLUMNS: columns that say what an MRD subset selects on; inside a subset on one of them, none
#   (nor the subset column itself) may be a trait or a covariate, so a launch that would is refused.  [column names]
SUBSET_LINKED_COLUMNS = mrd,mrd_level
# Study-cohort sensitivity (replaces env-MR-MEGA as the planned sensitivity, 17 Nov 2026): the MR-MEGA
# lead SNPs refitted in the whole cohort (POOLED: cohort PCs) with study_group as a categorical
# covariate (reference: the largest study group, @largest), a proxy for genotyping chip and treatment era; the other covariates
# as in the main model. Write the SNP list first (docs/WORKFLOW.md, "Study-cohort sensitivity"):
#     bin/mrmega_lead_snps.sh --output results/mrmega/lead_snps.txt results/mrmega/efs/efs.mrmega.result results/mrmega/os/os.mrmega.result
#     ARM=study_cohort slurm/submit_full_pipeline.sh
arm.study_cohort = STRATA_LIST=POOLED VARIANT_INCLUDE=results/mrmega/lead_snps.txt FACTOR_COVARIATES=cyto_risk:favorable,study_group:@largest
# LEAD_SNP_P: MR-MEGA association P below which bin/mrmega_lead_snps.sh lists a variant (one lead per
#   500 kb). This study: the suggestive threshold 1e-5 rather than genome-wide 5e-8, so the study-cohort
#   check also covers the nominally significant loci (18 Nov 2026).  [a number > 0]
LEAD_SNP_P = 1e-5

## The bundle checker (bin/check_handoff_bundle.sh) ----------------------------
# STUDY_GROUP_COL: the column naming each sample's study (the MR-MEGA study).  [a column name]
STUDY_GROUP_COL = study_group
# EOI_PROTOCOL_DAY: group:day -- every sample of these study groups must have this day in EOI_COL
#   (COG and St. Jude: the protocol day). Groups not listed (REDIAL) keep each child's own day,
#   which must be >= 0 or NA.  [group:number,...]
EOI_PROTOCOL_DAY = COG_P9900:29,COG_AALL0232_0331:29,SJ_TXV:46
# COLUMN_LEVELS: column:value/value/... -- the only values (besides NA) a column may hold.  [column:values,...]
COLUMN_LEVELS = mrd_level:1/2/3/4,cyto_risk:favorable/unfavorable/neutral,study_group:COG_P9900/COG_AALL0232_0331/SJ_TXV/REDIAL
# REQUIRED_COLUMNS: further columns the bundle must carry (never blank; NA allowed): protocol is
#   env-MR-MEGA's covariate (optional analysis, docs/ENV_MRMEGA.md).  [column names]
REQUIRED_COLUMNS = protocol,study_group
